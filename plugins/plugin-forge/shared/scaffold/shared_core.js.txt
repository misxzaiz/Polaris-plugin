/** 示例：设置规范化（非法输入回落默认值，绝不抛错） */
function normalizeSettings(raw) {
  const r = (raw && typeof raw === 'object') ? raw : {};
  return { example: r.example === true };
}

/** 示例：把业务规则写成纯函数 —— 输入→输出，不碰 kv/fetch/fs */
function planSomething(items, settings, now) {
  if (!Array.isArray(items)) return [];
  return items.filter((it) => it && it.enabled !== false);
}

module.exports = { normalizeSettings, planSomething };
